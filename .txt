    <div class="p-4 sticky top-0 bg-gray-50 z-10 shadow-sm">
        <input type="text" id="search-bar" oninput="filterProducts()" placeholder="🔍 ابحث عن وجبة أو منتج..." class="w-full p-3 border border-gray-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-green-500 bg-white shadow-sm">
        <div id="categories-container" class="flex gap-2 overflow-x-auto mt-3 pb-1">
            <button onclick="filterCategory('كل')" class="category-btn active px-4 py-1.5 text-sm rounded-full bg-white border border-gray-300 whitespace-nowrap shadow-sm">الكل</button>
        </div>
    </div>

    <div id="products-list" class="px-4 grid grid-cols-1 gap-3"></div>

    <div class="fixed bottom-0 left-0 right-0 bg-white shadow-[0_-5px_20px_rgba(0,0,0,0.05)] p-4 rounded-t-2xl z-20 max-w-md mx-auto border-t border-gray-100">
        <div class="flex justify-between items-center mb-3">
            <span class="font-bold text-gray-700">🛒 السلة (<span id="cart-count">0</span>)</span>
            <span id="cart-total" class="font-extrabold text-green-600 text-lg">0.00 $</span>
        </div>
        <div class="space-y-2 mb-3">
            <input type="text" id="client-name" placeholder=" اسم الزبون..." class="w-full p-2.5 text-sm border border-gray-200 rounded-lg bg-gray-50">
            <input type="text" id="client-address" placeholder=" العنوان بالتفصيل (دمشق / صحنايا)..." class="w-full p-2.5 text-sm border border-gray-200 rounded-lg bg-gray-50">
        </div>
        <button onclick="sendOrder()" class="w-full bg-green-500 hover:bg-green-600 text-white font-bold py-3 rounded-xl shadow-md transition-all">
            🚀 إرسال الطلب عبر الواتساب
        </button>
    </div>

    <script>
        const products = [
            { id: 1, name: "سلطة دجاج سانتا فيه", price: 8.49, category: "وجبات", image: "https://images.unsplash.com/photo-1546069901-ba9599a7e63c?w=150" },
            { id: 2, name: "برجر لحم سوبريم", price: 10.99, category: "وجبات", image: "https://images.unsplash.com/photo-1568901346375-23c9450c58cd?w=150" },
            { id: 3, name: "أصابع موزاريلا", price: 5.99, category: "مقبلات", image: "https://images.unsplash.com/photo-1531749668029-2db88e4b76ce?w=150" },
            { id: 4, name: "بطاطا مقلية مقرمشة", price: 3.50, category: "مقبلات", image: "https://images.unsplash.com/photo-1573080496219-bb080dd4f877?w=150" },
            { id: 5, name: "كوكا كولا", price: 1.50, category: "مشروبات", image: "https://images.unsplash.com/photo-1622483767028-3f66f32aef97?w=150" }
        ];

        const restaurantPhone = "963954573138"; 
        let cart = {};
        let currentCategory = "كل";

        function init() { renderCategories(); renderProducts(); }
        function renderCategories() {
            const cats = ["كل", ...new Set(products.map(p => p.category))];
            document.getElementById('categories-container').innerHTML = cats.map(c => `
                <button onclick="filterCategory('${c}')" class="category-btn ${c === currentCategory ? 'active' : ''} px-4 py-1.5 text-sm rounded-full bg-white border border-gray-300">${c}</button>
            `).join('');
        }
        function renderProducts(keyword = "") {
            const list = document.getElementById('products-list');
            const filtered = products.filter(p => (currentCategory === "كل" || p.category === currentCategory) && p.name.toLowerCase().includes(keyword.toLowerCase()));
            if(!filtered.length) { list.innerHTML = `<p class="text-center text-gray-400 py-8">لا توجد نتائج</p>`; return; }
            list.innerHTML = filtered.map(p => `
                <div class="bg-white p-3 rounded-xl shadow-sm border flex items-center justify-between">
                    <div class="flex items-center gap-3">
                        <img src="${p.image}" class="w-16 h-16 rounded-lg object-cover">
                        <div><h3 class="font-bold text-sm">${p.name}</h3><p class="text-green-600 text-xs mt-1">${p.price.toFixed(2)} $</p></div>
                    </div>
                    <div class="flex items-center gap-2">
                        <button onclick="changeQty(${p.id}, -1)" class="w-8 h-8 rounded-full bg-gray-100">-</button>
                        <span id="qty-${p.id}" class="font-bold text-sm w-4 text-center">0</span>
                        <button onclick="changeQty(${p.id}, 1)" class="w-8 h-8 rounded-full bg-green-500 text-white">+</button>
                    </div>
                </div>
            `).join('');
            Object.keys(cart).forEach(id => { if(document.getElementById(`qty-${id}`)) document.getElementById(`qty-${id}`).innerText = cart[id].qty; });
        }
        function changeQty(id, change) {
            if(!cart[id]) cart[id] = { ...products.find(p => p.id === id), qty: 0 };
            cart[id].qty += change;
            if(cart[id].qty <= 0) delete cart[id];
            if(document.getElementById(`qty-${id}`)) document.getElementById(`qty-${id}`).innerText = cart[id]?.qty || 0;
            updateCart();
        }
        function filterCategory(cat) { currentCategory = cat; renderCategories(); renderProducts(document.getElementById('search-bar').value); }
        function filterProducts() { renderProducts(document.getElementById('search-bar').value); }
        function updateCart() {
            let count = 0, total = 0;
            Object.values(cart).forEach(i => { count += i.qty; total += (i.price * i.qty); });
            document.getElementById('cart-count').innerText = count;
            document.getElementById('cart-total').innerText = total.toFixed(2) + " $";
        }
        function sendOrder() {
            const name = document.getElementById('client-name').value.trim();
            const address = document.getElementById('client-address').value.trim();
            if(!name || !address || !Object.keys(cart).length) { alert('أدخل الاسم والعنوان والطلبات أولاً'); return; }
            let text = `*طلب جديد* 📋\n\n*الزبون:* ${name}\n*العنوان:* ${address}\n-------------------------\n`;
            let total = 0;
            Object.values(cart).forEach(i => { total += (i.price * i.qty); text += `• ${i.name} (${i.qty}) -> ${(i.price * i.qty).toFixed(2)} $\n`; });
            text += `-------------------------\n*الإجمالي:* ${total.toFixed(2)} $`;
            window.open(`https://wa.me/${restaurantPhone}?text=${encodeURIComponent(text)}`, '_blank');
        }
        window.onload = init;
    </script>
